pragma solidity ^0.8.20;
contract StudentManagement{
    struct Student{
        uint256 stud_id;
        string stud_name;
        string depart;
    }


    Student[] public students;
    function addStudent(uint256 stud_id,string memory stud_name,string memory depart) public {
        Student memory stud_data = Student(stud_id,stud_name,depart);
        students.push(stud_data);
    }


    function getStudent(uint256 stud_id) public view returns (string memory,string memory){
        for(uint256 i=0;i<students.length;i++){
            if(students[i].stud_id== stud_id){
                return (students[i].stud_name,students[i].depart);
            }
        }
        revert("Student Not found");
    }
}
